// Writes the parked state in this folder, with production's own code.
//
// How to run it (it is not part of this branch's test sets: it imports files that
// exist only at that commit):
//
//   git worktree add --detach <dir> c388d81d        # main at release 0.12.13
//   cd <dir> && corepack pnpm install --frozen-lockfile --prefer-offline --ignore-scripts
//   cp <this file> apps/api/src/services/movement_engine/__test__/generate.integration.test.ts
//   createdb <scratch>                              # empty: globalSetup migrates it with that commit's migrations
//   cd apps/api && FIXTURE_DIR=<this folder> DATABASE_URL=<scratch> DATABASE_URL_TEST=<scratch> \
//     corepack pnpm exec jest --config src/test/jest-integration.config.ts generate.integration
//
// Seeded by hand here, not by production's save: the team, movement and trigger
// rows (`seed` below: kind manual, config {}), so a listen's real row shape is not
// evidence in these files. The movement_version row is production's own mint. Every
// other row was written by that commit's run host, engine (0.12.13, or its TypeScript
// engine for the legacy pin) and modules, except that `suspendAt` stands in for the
// cost meter in the limit fixture. `trigger_run.engine` is null throughout because
// this process never installs production's engine recorder. The dumps are whole rows,
// read back with `to_jsonb`, so a column production wrote and this branch ignores is
// kept. Each file repeats this under `generated`.

const TABLES = ['movement', 'movement_version', 'trigger', 'trigger_run', 'parked_run', 'join_pending', 'join_branch_export', 'adapter_await', 'callback', 'park_library_source'] as const;

async function dump(name: string, runId: string, movementId: string, triggerId: string): Promise<void> {
  const db = getAutomationsQb(['trigger_run']);
  const rows: Record<string, unknown[]> = {};
  const scopes: Record<(typeof TABLES)[number], ReturnType<typeof sql>> = {
    movement: sql`id = ${movementId}::uuid`,
    movement_version: sql`movement_id = ${movementId}::uuid`,
    trigger: sql`id = ${triggerId}::uuid`,
    trigger_run: sql`id = ${runId}::uuid`,
    parked_run: sql`run_id = ${runId}::uuid`,
    join_pending: sql`run_id = ${runId}::uuid`,
    join_branch_export: sql`run_id = ${runId}::uuid`,
    adapter_await: sql`run_id = ${runId}::uuid`,
    callback: sql`run_id = ${runId}::uuid`,
    park_library_source: sql`team_id = ${TEAM}::uuid`,
  };
  for (const table of TABLES) {
    const { rows: found } = await sql<{ row: unknown }>`SELECT to_jsonb(t) AS row FROM ${sql.table(`automations.${table}`)} t WHERE ${scopes[table]}`.execute(db);
    rows[table] = found.map((r) => r.row);
  }
  const { rows: work } = await sql<{ row: unknown }>`SELECT to_jsonb(w) AS row FROM automations.work w WHERE team_id = ${TEAM} AND (payload->>'runId' = ${runId} OR kind IN ('wake_await', 'fire_callback'))`.execute(db);
  rows.work = work.map((r) => r.row);
  const { rows: asks } = await sql<{ row: unknown }>`SELECT to_jsonb(a) AS row FROM asks.ask a WHERE team_id = ${TEAM}::uuid`.execute(db);
  rows.ask = asks.map((r) => r.row);
  writeFileSync(join(OUT, `${name}.json`), `${JSON.stringify({ writtenBy: 'c388d81d (0.12.13)', scenario: name, generated: generatedNote(name), runId, rows }, null, 2)}\n`);
}

const DISPATCH = "production's dispatchTriggerByIdEvent, through its run host and the 0.12.13 engine";
const LEGACY_DISPATCH = "production's dispatchTriggerByIdEvent, through its run host and its TypeScript engine for legacy pins";
const WRITERS: Record<string, string> = {
  timer: DISPATCH,
  ask: DISPATCH,
  callback: DISPATCH,
  race: DISPATCH,
  'callback-tapped': `${DISPATCH}, then production's fireCallback for the tap (the fire_callback and wake_await rows)`,
  'ask-answered': `${DISPATCH}, then production's answerAskById and its settled notifier for the answer (the wake_await row)`,
  limit: "production's TriggerRunRecorder, runMovement with the engine's suspendAt input standing in for the cost meter, its park sink, and markRunLimitPaused; no dispatch",
  'legacy-4-timer': LEGACY_DISPATCH,
  'legacy-4-ask': LEGACY_DISPATCH,
};

function generatedNote(name: string) {
  return {
    by: 'generate.integration.test.ts.txt in this folder, in a worktree at c388d81d, on a database that commit migrated',
    seededByHand: "the movement and trigger rows (the generator's own inserts, not production's save: kind manual, config {}, so a listen's real shape is not evidence here)",
    writtenByProduction: `movement_version by production's mintMovementVersionIfChanged; every other row by ${WRITERS[name]}`,
    notEvidence: "trigger_run.engine: the generator's process never installed production's engine recorder",
  };
}

function eventFor(triggerId: string, name: string) {
  return {
    pipelineInputId: `trigger:${triggerId}`,
    adapterType: 'manual',
    triggerType: 'webhook',
    payload: { firedAt: '2026-10-30T09:00:00.000Z', text: `written before the park: ${name}` },
    occurredAt: '2026-10-30T09:00:00.000Z',
  };
}

async function seed(name: string): Promise<{ movementId: string; triggerId: string; triggerName: string; versionId: string }> {
  const program = PROGRAMS[name]!;
  await sql`DELETE FROM asks.ask WHERE team_id = ${TEAM}::uuid`.execute(getAutomationsQb(['trigger_run']));
  await sql`DELETE FROM automations.work WHERE team_id = ${TEAM}`.execute(getAutomationsQb(['trigger_run']));
  const movementId = randomUUID();
  await getAutomationsQb(['movement'])
    .insertInto('movement')
    .values({ id: movementId, team_id: TEAM, name: `fixture_${name.replace(/-/g, '_')}_${movementId.slice(0, 8)}`, source: program.source, language_version: program.pin } as never)
    .execute();
  const { versionId } = await mintMovementVersionIfChanged({ teamId: TEAM as never, movementId: movementId as never, source: program.source, languageVersion: program.pin });
  const triggerId = randomUUID();
  await getAutomationsQb(['trigger'])
    .insertInto('trigger')
    .values({ id: triggerId, team_id: TEAM, name: `movement/fixture/${program.fired}`, kind: 'manual', movement_id: movementId, fired_movement_name: program.fired, config: {} } as never)
    .execute();
  return { movementId, triggerId, triggerName: `movement/fixture/${program.fired}`, versionId };
}

async function fire(name: string): Promise<{ runId: string; movementId: string; triggerId: string }> {
  const { movementId, triggerId } = await seed(name);
  const event = eventFor(triggerId, name);
  await dispatchTriggerByIdEvent({ teamId: TEAM as never, triggerId: triggerId as never, event } as never);
  const run = await getAutomationsQb(['trigger_run']).selectFrom('trigger_run').where('trigger_id', '=', triggerId).select(['id', 'status', 'failure_reason']).executeTakeFirstOrThrow();
  if (run.status !== 'parked') throw new Error(`${name}: the run is ${run.status}: ${String(run.failure_reason)}`);
  return { runId: run.id, movementId, triggerId };
}

describe('production writes its parked state', () => {
  beforeAll(async () => {
    mkdirSync(OUT, { recursive: true });
    await sql`DELETE FROM automations.trigger_run WHERE team_id = ${TEAM}`.execute(getAutomationsQb(['trigger_run']));
    await sql`DELETE FROM automations.trigger WHERE team_id = ${TEAM}::uuid`.execute(getAutomationsQb(['trigger_run']));
    await sql`DELETE FROM automations.movement WHERE team_id = ${TEAM}::uuid`.execute(getAutomationsQb(['trigger_run']));
    await getCoreQb(['team'])
      .insertInto('team')
      .values({ id: TEAM, name: 'fixture' } as never)
      .onConflict((oc) => oc.column('id').doNothing())
      .execute();
  });

  it.each(Object.keys(PROGRAMS))('%s', async (name) => {
    const fired = await fire(name);
    await dump(name, fired.runId, fired.movementId, fired.triggerId);
  });

  // What the database holds in the moment a deploy lands on a person's tap: the
  // call is on the ledger and its rows are queued, and no drain has run them.
  it('a callback tapped, its body queued', async () => {
    const fired = await fire('callback');
    const callback = await getAutomationsQb(['callback']).selectFrom('callback').where('run_id', '=', fired.runId as never).select('id').executeTakeFirstOrThrow();
    const outcome = await fireCallback({ id: callback.id, values: {} });
    if (outcome.kind !== 'recorded') throw new Error(`the tap came to ${outcome.kind}`);
    await dump('callback-tapped', fired.runId, fired.movementId, fired.triggerId);
  });

  it('an ask answered, its wake queued', async () => {
    notifyEngineOfSettledAsks();
    const fired = await fire('ask');
    const ask = await getAsksQb(['ask']).selectFrom('ask').where('team_id', '=', TEAM as never).select('id').executeTakeFirstOrThrow();
    const outcome = await answerAskById(ask.id, { Budget: '10' });
    if (outcome.kind !== 'answered') throw new Error(`the answer came to ${outcome.kind}`);
    await dump('ask-answered', fired.runId, fired.movementId, fired.triggerId);
  });

  // A flow suspended at a limit, written by the engine's own suspension and
  // production's sink. The cost meter itself is stood in for by `suspendAt`, the
  // engine's input for "suspend before this statement", which is what meeting the
  // cap does; the pause on the run row is production's own call.
  it('a run paused at its cost limit', async () => {
    const name = 'limit';
    const program = {
      fired: 'capped',
      source: `import { \`@lf/ask\` as ask } from adapters
qa = connect ask
function capped(run: <manual-[:Invocation]->>) {
  t = run.Text
  n = write qa-[:Form]-> { Prompt: t, Fields: ["after"] }
  return n
}
listen to manual fire capped
`,
    };
    PROGRAMS[name] = { pin: '0.12.8', ...program };
    const seeded = await seed(name);
    const event = eventFor(seeded.triggerId, name);
    const recorder = new TriggerRunRecorder({ teamId: TEAM as never, triggerId: seeded.triggerId, triggerName: seeded.triggerName, triggerType: 'webhook', triggerEvent: event as never, movementVersionId: seeded.versionId });
    await recorder.ensureStarted();
    const catalog = await runCatalogSnapshotForTeam(TEAM as never, { source: program.source, languageVersion: '0.12.8', imports: 'shipped' });
    const result = await runMovement({
      source: program.source,
      languageVersion: '0.12.8',
      movementName: program.fired,
      event: event as never,
      teamId: TEAM as never,
      firedAt: recorder.firedAt.toISOString(),
      catalog: catalog.snapshot,
      credentialIds: catalog.credentialIds,
      runId: recorder.triggerRunId,
      suspendAt: ['s1'],
      host: productionEngineHost({ teamId: TEAM as never, triggerRunId: recorder.triggerRunId, parksInto: recorder }),
    });
    if (result.parked !== true) throw new Error('the segment did not suspend');
    await markRunLimitPaused({ runId: recorder.triggerRunId, teamId: TEAM as never, triggerId: seeded.triggerId, pause: { limit: 'cost', capMicrodollars: 1000, spentMicrodollars: 2000 } });
    await dump(name, recorder.triggerRunId, seeded.movementId, seeded.triggerId);
  });
});
